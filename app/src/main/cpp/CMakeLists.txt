cmake_minimum_required(VERSION 3.22.1)
project(tabletalk LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set_property(GLOBAL PROPERTY JOB_POOLS compile_pool=2 link_pool=1)
set(CMAKE_JOB_POOL_COMPILE compile_pool)
set(CMAKE_JOB_POOL_LINK link_pool)
# Inference kernels must stay optimized even in signed Android debug APKs.
add_compile_options(-O3)

get_filename_component(WHISPER_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../../../third_party/whisper.cpp" ABSOLUTE)
if(NOT EXISTS "${WHISPER_DIR}/CMakeLists.txt")
    message(FATAL_ERROR "Run scripts/bootstrap.sh to fetch the pinned whisper.cpp source.")
endif()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(WHISPER_CURL OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_CPU_ALL_VARIANTS OFF CACHE BOOL "" FORCE)
set(GGML_CPU_ARM_ARCH "armv8-a" CACHE STRING "" FORCE)
# ARMv8-A baseline works on both Snapdragon 821 (G6) and 845 (G7).
# Do not globally require ARMv8.2 FP16, which would exclude the G6.
if(ANDROID_ABI STREQUAL "arm64-v8a")
    add_compile_options(-march=armv8-a)
endif()
add_subdirectory("${WHISPER_DIR}" "${CMAKE_BINARY_DIR}/whisper")
add_library(tabletalk-whisper SHARED whisper_jni.cpp)
target_link_libraries(tabletalk-whisper PRIVATE whisper)
